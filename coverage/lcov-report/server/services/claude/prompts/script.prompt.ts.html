export function buildScriptUserMessage(
  topicData: string,
  variant: 'A' | 'B',
  brandContext?: string
): string {
  const variantGuide = variant === 'A'
    ? `【A版本：情感驱动型】
重点：强情绪共鸣、具体生活场景、痛点代入感强
Hook策略：优先用问题崩塌型或理发店反转型
节奏：情绪铺垫更长（3-10s），转化收口要温和
CTA：用理解式语气（"如果你也..."）`
    : `【B版本：理性驱动型】
重点：数据背书、成分/特证、性价比锚点、套组逻辑
Hook策略：优先用值感反差型或测评对比型
节奏：卖点输出更集中（直接在10s内完成产品介绍）
CTA：用认知升级语气（"发膜这种东西，真别按一次买"）`
 
  return `为以下选题创作${variant}版本投流脚本。
 
${variantGuide}
 
${brandContext ? `【话术参考与产品卖点（IMPORTANT - 必须参考）】
以下是品牌提供的官方话术参考、产品卖点和违禁词指南，生成脚本时：
1. 优先使用这些官方卖点表述
2. 严格遵守违禁词规范
3. 产品名称、规格、功效必须与参考文件一致
 
${brandContext}
 
---
` : ''}
 
【选题信息】
${topicData}
 
**脚本生成要求**：
1. 用<script>标签包裹完整JSON脚本
2. 确保逻辑连贯：每个segment之间要有自然过渡，形成完整说服链路
3. 口播文案完整且自然（fullVoiceover字段）
4. 每个segment都有具体镜头描述（shot字段）
5. 混剪策略完整（mixCutStrategy字段含素材池/镜头顺序/字幕节奏）
6. A/B版本形成明显差异化
7. ${brandContext ? '严格遵守话术参考和违禁词指南' : ''}
 
**逻辑连贯性检查清单**（生成后自我检查）：
- [ ] hook是否自然引出问题？
- [ ] 问题放大是否承接hook？
- [ ] 产品出场是否作为问题的解决方案？
- [ ] 卖点是否支撑产品能解决问题？
- [ ] 证明是否验证卖点真实有效？
- [ ] CTA是否自然收口，给出明确行动指引？`
}

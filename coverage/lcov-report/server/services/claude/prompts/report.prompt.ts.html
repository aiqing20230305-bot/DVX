export interface ReportSection {
  id: string
  title: string
  content: string
  subsections?: Array<{ title: string; content: string }>
}
 
export interface StrategyReport {
  brand: string
  product: string
  deliveryDate: string
  coreFormula: string         // 核心一句话（总策略公式）
  sections: ReportSection[]

    rank: number
    name: string
    tag: string
    direction: string
    targetPersona: string
    core: string
    cta: string
  }>
  scripts: Array<{
    name: string
    positioning: string
    hook: string
    structure: string[]
    scenes: string[]

  }>

    subtitleStyle: string[]

}

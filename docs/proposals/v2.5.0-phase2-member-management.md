# v2.5.0 Phase 2: 项目成员管理 - 实施方案

**版本**: v2.5.0-phase2  
**预计周期**: 4天 (Day 8-11)  
**前置条件**: Phase 1 用户管理系统已完成  
**目标**: 实现多人协作的项目成员管理和权限控制

---

## 📋 Phase 2 范围

### 核心功能

1. **项目成员管理**
   - 邀请成员加入项目
   - 查看项目成员列表
   - 修改成员角色（owner/editor/viewer）
   - 移除项目成员
   
2. **权限控制 (RBAC)**
   - owner: 完全控制（管理成员、修改项目、删除项目）
   - editor: 编辑权限（创建/修改洞察、选题、脚本，不能管理成员）
   - viewer: 只读权限（查看内容，不能修改）
   
3. **API权限保护**
   - 所有项目相关API添加成员验证
   - 基于角色的操作权限检查
   - 项目owner自动创建

### 不包含

- 评论功能（Phase 3）
- 审批流程（Phase 4）
- 邮件邀请通知（后期优化）

## 🗄️ 数据库设计

### 新增表：project_members

```sql
CREATE TABLE IF NOT EXISTS project_members (
  id TEXT PRIMARY KEY,
  project_id TEXT NOT NULL,
  user_id TEXT NOT NULL,
  role TEXT NOT NULL CHECK (role IN ('owner', 'editor', 'viewer')),
  invited_by TEXT,
  joined_at INTEGER NOT NULL,
  created_at INTEGER NOT NULL,
  updated_at INTEGER NOT NULL,
  FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  FOREIGN KEY (invited_by) REFERENCES users(id),
  UNIQUE(project_id, user_id)
);

CREATE INDEX idx_project_members_project_id ON project_members(project_id);
CREATE INDEX idx_project_members_user_id ON project_members(user_id);
```

### 修改现有表：projects

添加 `created_by` 字段：

```sql
ALTER TABLE projects ADD COLUMN created_by TEXT;
```

**数据迁移策略**：
- 现有项目的 `created_by` 设为 'system'（超级管理员账户）
- 或在首次部署时要求用户登录后认领项目

### 2. Middleware层 (server/middleware/)

#### permission.middleware.ts（新建）

```typescript
import { Request, Response, NextFunction } from 'express'
import { ProjectMemberRepository } from '../db/repositories/project-member.repo'

// 扩展Request类型，添加projectId和member信息
declare global {
  namespace Express {
    interface Request {
      projectId?: string
      member?: ProjectMember
    }
  }
}

// 验证用户是否是项目成员（任意角色）
export const requireProjectMember = async (req: Request, res: Response, next: NextFunction) => {
  const userId = req.user?.id
  const projectId = req.params.projectId || req.body.project_id
  
  if (!userId || !projectId) {
    return res.status(401).json({ error: 'Unauthorized' })
  }
  
  const memberRepo = new ProjectMemberRepository(req.db)
  const member = await memberRepo.getMember(projectId, userId)
  
  if (!member) {
    return res.status(403).json({ error: 'Not a member of this project' })
  }
  
  req.projectId = projectId
  req.member = member
  next()
}

// 验证用户是否有指定角色或更高权限
export const requireRole = (minRole: 'viewer' | 'editor' | 'owner') => {
  const roleHierarchy = { viewer: 0, editor: 1, owner: 2 }
  
  return async (req: Request, res: Response, next: NextFunction) => {
    const member = req.member
    
    if (!member) {
      return res.status(403).json({ error: 'Not a member of this project' })
    }
    
    const userRoleLevel = roleHierarchy[member.role]
    const requiredRoleLevel = roleHierarchy[minRole]
    
    if (userRoleLevel < requiredRoleLevel) {
      return res.status(403).json({ error: `Requires ${minRole} role or higher` })
    }
    
    next()
  }
}
```

### 3. API Routes (server/routes/)

#### project-members.route.ts（新建）

```typescript
import { Router } from 'express'
import { authMiddleware } from '../middleware/auth.middleware'
import { requireProjectMember, requireRole } from '../middleware/permission.middleware'

const router = Router()

/**
 * GET /api/projects/:projectId/members
 * 获取项目成员列表
 * 权限：任意项目成员
 */
router.get(
  '/:projectId/members',
  authMiddleware,
  requireProjectMember,
  async (req, res) => {
    const memberRepo = new ProjectMemberRepository(req.db)
    const members = await memberRepo.getMembersByProject(req.projectId!)
    res.json({ members })
  }
)

/**
 * POST /api/projects/:projectId/members
 * 邀请新成员
 * 权限：owner或editor
 * Body: { email: string, role: 'editor' | 'viewer' }
 */
router.post(
  '/:projectId/members',
  authMiddleware,
  requireProjectMember,
  requireRole('editor'),
  async (req, res) => {
    const { email, role } = req.body
    const inviterId = req.user!.id
    
    // 查找被邀请用户
    const userRepo = new UserRepository(req.db)
    const invitedUser = await userRepo.findByEmail(email)
    
    if (!invitedUser) {
      return res.status(404).json({ error: 'User not found' })
    }
    
    // 检查是否已是成员
    const memberRepo = new ProjectMemberRepository(req.db)
    const existing = await memberRepo.getMember(req.projectId!, invitedUser.id)
    
    if (existing) {
      return res.status(400).json({ error: 'User is already a member' })
    }
    
    // 添加成员
    const member = await memberRepo.addMember(
      req.projectId!,
      invitedUser.id,
      role,
      inviterId
    )
    
    // 记录时间线
    await timelineRepo.create({
      project_id: req.projectId!,
      event_type: 'member_added',
      details: JSON.stringify({
        member_name: invitedUser.name,
        role,
        invited_by: req.user!.name
      })
    })
    
    res.json({ member })
  }
)

/**
 * PUT /api/projects/:projectId/members/:memberId
 * 修改成员角色
 * 权限：仅owner
 * Body: { role: 'owner' | 'editor' | 'viewer' }
 */
router.put(
  '/:projectId/members/:memberId',
  authMiddleware,
  requireProjectMember,
  requireRole('owner'),
  async (req, res) => {
    const { memberId } = req.params
    const { role } = req.body
    
    const memberRepo = new ProjectMemberRepository(req.db)
    await memberRepo.updateMemberRole(memberId, role)
    
    // 记录时间线
    await timelineRepo.create({
      project_id: req.projectId!,
      event_type: 'member_role_updated',
      details: JSON.stringify({
        member_id: memberId,
        new_role: role,
        updated_by: req.user!.name
      })
    })
    
    res.json({ message: 'Member role updated' })
  }
)

/**
 * DELETE /api/projects/:projectId/members/:memberId
 * 移除成员
 * 权限：owner（不能移除自己）
 */
router.delete(
  '/:projectId/members/:memberId',
  authMiddleware,
  requireProjectMember,
  requireRole('owner'),
  async (req, res) => {
    const { memberId } = req.params
    
    // 不能移除自己
    const memberRepo = new ProjectMemberRepository(req.db)
    const member = await memberRepo.getMember(req.projectId!, req.user!.id)
    if (member?.id === memberId) {
      return res.status(400).json({ error: 'Cannot remove yourself' })
    }
    
    await memberRepo.removeMember(memberId)
    
    // 记录时间线
    await timelineRepo.create({
      project_id: req.projectId!,
      event_type: 'member_removed',
      details: JSON.stringify({
        member_id: memberId,
        removed_by: req.user!.name
      })
    })
    
    res.json({ message: 'Member removed' })
  }
)

export default router
```

```typescript
import { create } from 'zustand'

interface Member {
  id: string
  user: {
    id: string
    name: string
    email: string
    avatar?: string
  }
  role: 'owner' | 'editor' | 'viewer'
  joined_at: number
}

interface MemberStore {
  members: Member[]
  loading: boolean
  error: string | null
  
  // Actions
  fetchMembers: (projectId: string) => Promise<void>
  inviteMember: (projectId: string, email: string, role: string) => Promise<void>
  updateMemberRole: (projectId: string, memberId: string, role: string) => Promise<void>
  removeMember: (projectId: string, memberId: string) => Promise<void>
  reset: () => void
}

export const useMemberStore = create<MemberStore>((set, get) => ({
  members: [],
  loading: false,
  error: null,
  
  fetchMembers: async (projectId) => {
    set({ loading: true, error: null })
    try {
      const res = await fetch(`/api/projects/${projectId}/members`)
      if (!res.ok) throw new Error('Failed to fetch members')
      const data = await res.json()
      set({ members: data.members, loading: false })
    } catch (error) {
      set({ error: error.message, loading: false })
    }
  },
  
  inviteMember: async (projectId, email, role) => {
    try {
      const res = await fetch(`/api/projects/${projectId}/members`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, role })
      })
      if (!res.ok) throw new Error('Failed to invite member')
      await get().fetchMembers(projectId) // Refresh list
      toast.success('成员已邀请')
    } catch (error) {
      toast.error(error.message)
      throw error
    }
  },
  
  updateMemberRole: async (projectId, memberId, role) => {
    try {
      const res = await fetch(`/api/projects/${projectId}/members/${memberId}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ role })
      })
      if (!res.ok) throw new Error('Failed to update role')
      await get().fetchMembers(projectId)
      toast.success('角色已更新')
    } catch (error) {
      toast.error(error.message)
      throw error
    }
  },
  
  removeMember: async (projectId, memberId) => {
    try {
      const res = await fetch(`/api/projects/${projectId}/members/${memberId}`, {
        method: 'DELETE'
      })
      if (!res.ok) throw new Error('Failed to remove member')
      await get().fetchMembers(projectId)
      toast.success('成员已移除')
    } catch (error) {
      toast.error(error.message)
      throw error
    }
  },
  
  reset: () => set({ members: [], loading: false, error: null })
}))
```

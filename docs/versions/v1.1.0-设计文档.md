### Task 2: 安全配置优化

**目标**: 增强安全性，遵循最佳实践

**当前问题**:
- CORS配置过于宽松（`origin: '*'`）
- 环境变量验证不完善
- 缺少安全头配置

**优化方案**:

#### 2.1 CORS配置优化

**当前代码** (`server/index.ts`):
```typescript
app.use(cors({ origin: '*' }))
```

**优化后**:
```typescript
const allowedOrigins = process.env.ALLOWED_ORIGINS?.split(',') || [
  'http://localhost:5173', // 开发环境
  'http://localhost:3001', // 生产环境（同域）
]

app.use(cors({
  origin: (origin, callback) => {
    // 允许同域请求（无origin）
    if (!origin) return callback(null, true)
    
    // 检查是否在白名单中
    if (allowedOrigins.includes(origin)) {
      callback(null, true)
    } else {
      callback(new Error('Not allowed by CORS'))
    }
  },
  credentials: true,
  methods: ['GET', 'POST', 'PUT', 'DELETE'],
  allowedHeaders: ['Content-Type', 'Authorization'],
}))
```

**环境变量更新** (`.env.example`):
```bash
# CORS允许的来源（逗号分隔）
ALLOWED_ORIGINS=http://localhost:5173,http://localhost:3001
```

```typescript
export function validateEnv(): void {
  const required = ['ANTHROPIC_API_KEY']
  const missing = required.filter(key => !process.env[key])
  
  if (missing.length > 0) {
    console.error('❌ 缺少必需的环境变量:', missing.join(', '))
    console.error('请参考 .env.example 配置环境变量')
    process.exit(1)
  }
  
  // API Key格式验证
  if (process.env.ANTHROPIC_API_KEY && !process.env.ANTHROPIC_API_KEY.startsWith('sk-ant-')) {
    console.warn('⚠️  ANTHROPIC_API_KEY 格式可能不正确')
  }
  
  console.log('✅ 环境变量验证通过')
}
```

**新增** (`server/types/errors.ts`):
```typescript
export enum ErrorCode {
  // 客户端错误 (4xx)
  VALIDATION_ERROR = 'VALIDATION_ERROR',
  RESOURCE_NOT_FOUND = 'RESOURCE_NOT_FOUND',
  UNAUTHORIZED = 'UNAUTHORIZED',
  
  // 服务器错误 (5xx)
  DATABASE_ERROR = 'DATABASE_ERROR',
  AI_API_ERROR = 'AI_API_ERROR',
  FILE_PROCESSING_ERROR = 'FILE_PROCESSING_ERROR',
}

export class DetailedAppError extends Error {
  constructor(
    public statusCode: number,
    public code: ErrorCode,
    public message: string,
    public details?: any,
    public suggestion?: string
  ) {
    super(message)
  }
}
```

#### 3.2 前端错误边界增强

**新增** (`src/components/ErrorBoundary.tsx`):
```typescript
import React, { Component, ReactNode } from 'react'

interface Props {
  children: ReactNode
  fallback?: ReactNode
}

interface State {
  hasError: boolean
  error?: Error
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props)
    this.state = { hasError: false }
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error }
  }

  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
    console.error('ErrorBoundary caught:', error, errorInfo)
  }

  render() {
    if (this.state.hasError) {
      return this.props.fallback || (
        <div className="error-fallback">
          <h2>出错了</h2>
          <p>{this.state.error?.message}</p>
          <button onClick={() => window.location.reload()}>
            刷新页面
          </button>
        </div>
      )
    }

    return this.props.children
  }
}
```

**集成** (`src/App.tsx`):
```typescript

  <Router>
    {/* ... */}
  </Router>
</ErrorBoundary>
```
